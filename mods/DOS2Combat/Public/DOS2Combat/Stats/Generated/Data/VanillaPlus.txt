-- Tavern Brawler for every attack ability: the attack's own ability modifier
-- is added a second time to the attack roll and to its damage.
--   melee weapons  -> best of Strength/Dexterity
--   ranged weapons -> Dexterity
--   unarmed        -> Strength (vanilla Tavern Brawler)
--   spell attacks  -> spellcasting ability (Int/Wis/Cha)
new entry "DOS2C_UniversalBrawler"
type "PassiveData"
data "DisplayName" "h8d82818ba122548c88203edd1df48e2b"
data "Description" "h64f6f3e2220c544f991c3d5b72d009ae"
data "Icon" "PassiveFeature_TavernBrawler"
data "Properties" "Highlighted"
data "Boosts" "RollBonus(MeleeWeaponAttack,max(0,max(StrengthModifier,DexterityModifier)));RollBonus(RangedWeaponAttack,max(0,DexterityModifier));RollBonus(MeleeUnarmedAttack,max(0,StrengthModifier));RollBonus(MeleeSpellAttack,max(0,SpellCastingAbilityModifier));RollBonus(RangedSpellAttack,max(0,SpellCastingAbilityModifier));IF(IsMeleeWeaponAttack()):DamageBonus(max(0,max(StrengthModifier,DexterityModifier)));IF(IsWeaponAttack() and IsRangedAttack()):DamageBonus(max(0,DexterityModifier));IF(IsMeleeUnarmedAttack()):DamageBonus(max(0,StrengthModifier));IF(IsSpell() and IsAttack()):DamageBonus(max(0,SpellCastingAbilityModifier))"

-- Applied to hostile combatants when they enter combat.

type "StatusData"
data "StatusType" "BOOST"
data "StackId" "DOS2C_ENEMY_HP"
data "Boosts" "IncreaseMaxHP(50%)"
data "StatusPropertyFlags" "DisableOverhead;DisableCombatlog;DisablePortraitIndicator;IgnoreResting"
